// ### O CRUD na tela: o que cada letra vira

C - Create  = cadastrar  -> formulário + fetch POST
R - Read    = listar     -> fetch GET + montar a tabela na tela
U - Update  = editar     -> botão Editar preenche o formulário + fetch PUT
D - Delete  = excluir    -> botão Excluir + fetch DELETE

Depois de CADA ação (cadastrar, editar, excluir) chamamos carregar() de novo para a tabela mostrar o que mudou.


// ### Ferramentas do DOM (como o JavaScript mexe na página)

document.getElementById('lista')   pega o elemento que tem id="lista"
.value                             lê (ou escreve) o que está dentro de um input
.textContent                       escreve texto simples em um elemento
.innerHTML                         escreve HTML dentro de um elemento (usado para montar as linhas da tabela)
array.forEach(item => { ... })     repete um bloco para cada item do array que veio da API
array.find(item => condição)       acha o primeiro item que combina (usado no Editar)
`texto ${variavel} texto`          template literal: crase + ${} coloca variável no meio do texto

Regra de ouro da lista: LIMPE antes (lista.innerHTML = '') e depois SOME com += dentro do forEach.
Se não limpar, cada recarga repete tudo.

// ### Listar (R): GET + forEach + innerHTML

// >>> FRONT/barbeiros.html
        async function carregar(busca) {
            const caminho = busca ? `/barbeiros?busca=${encodeURIComponent(busca)}` : '/barbeiros'
            barbeiros = await api(caminho)

            const lista = document.getElementById('lista')
            lista.innerHTML = ''

            barbeiros.forEach(barbeiro => {
                lista.innerHTML += `
                    <tr>
                        <td>${barbeiro.id}</td>
                        <td>${barbeiro.nome}</td>
                        <td>${barbeiro.email}</td>
                        <td>${barbeiro.telefone || ''}</td>
                        <td>
                            <button onclick="editar(${barbeiro.id})">Editar</button>
                            <button onclick="excluir(${barbeiro.id})">Excluir</button>
                        </td>
                    </tr>
                `
            })
        }
// <<<

// - api('/barbeiros') chama GET /barbeiros mandando o token (função do scripts.js).
// - O parâmetro busca é da Aula 9: se vier vazio, o caminho é só '/barbeiros' e lista todos.
// - ${barbeiro.telefone || ''} evita aparecer "null" na tela quando o telefone está vazio.

// ### Cadastrar (C) e também Salvar a edição (U): POST ou PUT

Se o input escondido id está vazio, é cadastro (POST). Se tem id, é edição (PUT).

// >>> FRONT/barbeiros.html
        async function salvar(event) {
            event.preventDefault()

            const id = document.getElementById('id').value
            const barbeiro = {
                nome: document.getElementById('nome').value,
                email: document.getElementById('email').value,
                cpf: document.getElementById('cpf').value,
                senha: document.getElementById('senha').value,
                telefone: document.getElementById('telefone').value
            }

            let dados
            if (id) {
                dados = await api(`/barbeiros/${id}`, 'PUT', barbeiro)
            } else {
                dados = await api('/barbeiros', 'POST', barbeiro)
            }

            alert(dados.message)
            limparFormulario()
            carregar()
        }
// <<<

// Na Aula 8 só existe o ramo do POST. O ramo do PUT (if (id)) entra na Aula 9.
// A API devolve { message: '...' }, e mostramos essa mensagem com alert.


// ### Excluir (D): DELETE com confirmação

// >>> FRONT/barbeiros.html
        async function excluir(id) {
            if (!confirm('Deseja excluir este barbeiro?')) return

            const dados = await api(`/barbeiros/${id}`, 'DELETE')
            alert(dados.message)
            carregar()
        }
// <<<

// confirm() abre a janelinha OK/Cancelar. Se cancelar, o return sai da função sem excluir.


// ### AULA 9: Busca por termo (critério 5.2)

O usuário digita o termo, confirma (botão Buscar ou Enter) e a tabela mostra só os registros que combinam.
O botão Limpar volta para a lista completa.
A parte da API é o ?busca= com ILIKE (veja #4_filtros_e_JOIN.txt).

// >>> FRONT/barbeiros.html
        function buscar(event) {
            event.preventDefault()
            carregar(document.getElementById('termo').value)
        }

        function limparBusca() {
            document.getElementById('termo').value = ''
            carregar()
        }
// <<<


// ### AULA 9: Editar (critério 5.5)

Passo a passo do que acontece:
1. O usuário clica em Editar na linha do barbeiro.
2. editar(id) acha o barbeiro na lista (find) e joga os dados nos inputs do formulário.
3. O input escondido id passa a ter o id do barbeiro, então o salvar() sabe que é PUT.
4. O usuário muda o que quiser e clica em "Salvar alterações".
5. A tabela recarrega e mostra o dado novo.

CPF e senha não voltam preenchidos (estão em hash, não dá para mostrar). Só são trocados se o usuário digitar de novo.

// >>> FRONT/barbeiros.html
        // Preenche o formulário com os dados do barbeiro escolhido
        function editar(id) {
            const barbeiro = barbeiros.find(b => b.id === id)

            document.getElementById('id').value = barbeiro.id
            document.getElementById('nome').value = barbeiro.nome
            document.getElementById('email').value = barbeiro.email
            document.getElementById('telefone').value = barbeiro.telefone || ''
            // CPF e senha estão criptografados: só preenche se quiser trocar
            document.getElementById('cpf').placeholder = 'Novo CPF (opcional)'
            document.getElementById('senha').placeholder = 'Nova senha (opcional)'

            document.getElementById('tituloForm').textContent = 'Editar barbeiro'
            document.getElementById('botaoSalvar').textContent = 'Salvar alterações'
        }

        function limparFormulario() {
            document.getElementById('formBarbeiro').reset()
            document.getElementById('id').value = ''
            document.getElementById('cpf').placeholder = 'CPF'
            document.getElementById('senha').placeholder = 'Senha'
            document.getElementById('tituloForm').textContent = 'Cadastrar barbeiro'
            document.getElementById('botaoSalvar').textContent = 'Cadastrar'
        }
// <<<

// ### Erros comuns

- A tabela repete as linhas a cada ação: faltou lista.innerHTML = '' antes do forEach.
- Aparece "undefined" na tabela: o nome do campo está diferente do que a API devolve (confira no Network, F12).
- O Editar não preenche nada: a função editar usa a lista "barbeiros"; ela só é preenchida dentro do carregar().
- O Editar/Excluir falha só no navegador: faltou PUT/DELETE em methods no CORS do server.js.
- O botão Cadastrar recarrega a página: faltou event.preventDefault() no começo da função.
- onclick="editar(${barbeiro.id})" sem as crases: o HTML montado em string precisa estar entre crases ( ` ), não aspas.
- Editou o barbeiro mas o CPF mudou "sozinho": digitou algo no campo CPF. Deixe vazio para manter o atual.
