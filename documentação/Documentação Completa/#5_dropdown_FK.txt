// ### A ideia

O caderno diz: "Ao cadastrar um novo cliente, o usuário deve associá-lo a um barbeiro."
No banco isso é a chave estrangeira cliente.barbeiro_id.
O usuário NÃO deve digitar o número do barbeiro. Ele escolhe o NOME numa lista (<select>)
e o sistema envia o ID escondido:

1. O front busca os barbeiros (GET /barbeiros).
2. Cada barbeiro vira uma <option>: o TEXTO é o nome e o VALUE é o id.
3. Ao cadastrar, o id escolhido vai no corpo do POST como barbeiro_id.
4. A API confere se esse barbeiro existe antes de gravar.


// ### Passo 1: API, rota POST /clientes

A listagem GET /clientes (com JOIN, para trazer o nome do barbeiro) está em #4_filtros_e_JOIN.txt.
Cole no server.js, junto das outras rotas:

// >>> server.js
servidor.post('/clientes', { preHandler: autenticar }, async (request, reply) => {
    const body = request.body

    if (!body || !body.nome || !body.barbeiro_id) {
        return reply.status(400).send({ message: 'Nome e barbeiro são obrigatórios!' })
    }

    const barbeiro = await sql.query('SELECT id FROM barbeiro WHERE id = $1', [body.barbeiro_id])
    if (barbeiro.rows.length === 0) {
        return reply.status(400).send({ message: 'Barbeiro informado não existe!' })
    }

    await sql.query(
        'INSERT INTO cliente (nome, telefone, estilo_corte, barbeiro_id) VALUES ($1, $2, $3, $4)',
        [body.nome, body.telefone, body.estilo_corte, body.barbeiro_id]
    )

    return reply.status(201).send({ message: 'Cliente cadastrado!' })
})
// <<<

// A conferência (SELECT id FROM barbeiro...) devolve uma mensagem clara. Sem ela, um barbeiro_id inexistente
// estoura um erro de chave estrangeira do Postgres e a API responde 500.
// (Se ainda não fez a função autenticar, tire o { preHandler: autenticar } por enquanto.)

// ### Passo 3: preencher o select com os barbeiros do banco

Cole DENTRO do <script>, abaixo do protegerPagina():

// >>> FRONT/clientes.html
        // Preenche o select com os barbeiros do banco (value = id, texto = nome)
        async function carregarBarbeiros() {
            const barbeiros = await api('/barbeiros')
            const select = document.getElementById('barbeiro_id')

            barbeiros.forEach((barbeiro) => {
                select.innerHTML += `<option value="${barbeiro.id}">${barbeiro.nome}</option>`
            })
        }
// <<<

// - Para cada barbeiro, somamos (+=) uma <option> dentro do select, igual ao que fazemos com a lista.
// - value="${barbeiro.id}" é o que vai para o banco (a chave estrangeira).
// - ${barbeiro.nome} é o que o usuário lê.
// - Esta função precisa ser chamada: veja o passo 5.


// ### Passo 4: listar os clientes, com o nome do barbeiro

Cole abaixo da função anterior:

// >>> FRONT/clientes.html
        async function carregarClientes() {
            const clientes = await api('/clientes')
            const lista = document.getElementById('lista')
            lista.innerHTML = ''

            clientes.forEach((cliente) => {
                lista.innerHTML += `
                <div class="item">
                    <b>${cliente.nome}</b> <br>
                    telefone: ${cliente.telefone || ''} <br>
                    estilo de corte: ${cliente.estilo_corte || ''} <br>
                    barbeiro: ${cliente.barbeiro_nome}
                </div>`
            })
        }
// <<<

// ${cliente.barbeiro_nome} vem do JOIN do GET /clientes (#4_filtros_e_JOIN.txt).


// ### Passo 5: cadastrar o cliente

Cole abaixo da função anterior. Se o usuário não escolher o barbeiro, o value do select é vazio
e bloqueamos o envio com um aviso:

// >>> FRONT/clientes.html
        async function cadastrar() {
            const cliente = {
                nome: document.getElementById('nome').value,
                telefone: document.getElementById('telefone').value,
                estilo_corte: document.getElementById('estilo_corte').value,
                barbeiro_id: document.getElementById('barbeiro_id').value
            }

            if (!cliente.barbeiro_id) {
                alert('Selecione um barbeiro!')
                return
            }

            const dados = await api('/clientes', 'POST', cliente)
            alert(dados.message)

            document.getElementById('nome').value = ''
            document.getElementById('telefone').value = ''
            document.getElementById('estilo_corte').value = ''
            document.getElementById('barbeiro_id').value = ''
            carregarClientes()
        }
// <<<

E, por último no script (acima do </script>), chame as duas funções que carregam a tela:

// >>> FRONT/clientes.html

// ### Erros comuns

- O select fica só com "selecione...": faltou chamar carregarBarbeiros() no final do script, ou o GET /barbeiros falhou
  (sem token, API parada). Veja o Network no F12.
- Cadastrou, mas o barbeiro aparece como "undefined": o campo vem do JOIN, confira o apelido barbeiro_nome no GET /clientes.
- "Barbeiro informado não existe!": o value da option está errado (use barbeiro.id, não o nome).
- O texto da option aparece cortado ou estranho: a option precisa de abertura e fechamento (<option ...>texto</option>).
